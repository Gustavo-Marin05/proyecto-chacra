js
(() => {
  const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  if (prefersReducedMotion) return;

  // 1. Parallax 3D en la escena isométrica del Hero
  const heroStage = document.getElementById('hero-stage');
  const isoScene = document.getElementById('iso-scene');

  if (heroStage && isoScene) {
    heroStage.addEventListener('mousemove', (e) => {
      const rect = heroStage.getBoundingClientRect();
      const x = (e.clientX - rect.left) / rect.width - 0.5;
      const y = (e.clientY - rect.top) / rect.height - 0.5;

      const rotX = 55 - y * 14;
      const rotZ = -45 + x * 18;
      isoScene.style.transform = `rotateX(${rotX.toFixed(2)}deg) rotateZ(${rotZ.toFixed(2)}deg)`;
    });

    heroStage.addEventListener('mouseleave', () => {
      isoScene.style.transform = 'rotateX(55deg) rotateZ(-45deg)';
    });
  }

  // 2. Inclinación 3D (hasta 10°) en tarjetas con capa interior elevada
  const cards = document.querySelectorAll('[data-tilt]');
  const MAX_TILT = 10;

  cards.forEach((card) => {
    card.addEventListener('mousemove', (e) => {
      const rect = card.getBoundingClientRect();
      const x = (e.clientX - rect.left) / rect.width - 0.5;
      const y = (e.clientY - rect.top) / rect.height - 0.5;

      const rotateX = (-y * 2 * MAX_TILT).toFixed(2);
      const rotateY = (x * 2 * MAX_TILT).toFixed(2);

      card.style.transform = `rotateX(${rotateX}deg) rotateY(${rotateY}deg)`;
    });

    card.addEventListener('mouseleave', () => {
      card.style.transform = 'rotateX(0deg) rotateY(0deg)';
    });
  });
})();
